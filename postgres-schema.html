Overtime Management PostgreSQL Schema

PostgreSQL Schema — Etihad Rail Overtime App

7 tables derived from the current TypeScript data model in mockData.ts and dataService.ts. OT hours are stored flat (not as nested JSON) for query performance.

Employee IDs remain string-keyed (e.g. EMP-1001) to match the existing mock data format and avoid a migration step.

-- ============================================================ -- Etihad Rail Overtime Management — PostgreSQL Schema -- Generated: 2026-07-07 -- ============================================================ -- ── Extensions ────────────────────────────────────────────── CREATE EXTENSION IF NOT EXISTS "pgcrypto"; -- ── Enums ──────────────────────────────────────────────────── CREATE TYPE user_role AS ENUM ('EMPLOYEE', 'MANAGER', 'HR'); CREATE TYPE manager_level AS ENUM ('L1', 'L2'); CREATE TYPE approval_status AS ENUM ('Pending', 'Approved', 'Rejected'); CREATE TYPE attendance_status AS ENUM ('Present', 'Weekend', 'Leave', 'Holiday'); -- ── grade_pay_scales ───────────────────────────────────────── CREATE TABLE grade_pay_scales ( grade VARCHAR(5) PRIMARY KEY, gross_monthly_pay NUMERIC(12,2) NOT NULL, updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- Seed: G1 = 12 000 AED, G2 = 11 000 … G12 = 1 000 INSERT INTO grade_pay_scales (grade, gross_monthly_pay) VALUES ('G1', 12000), ('G2', 11000), ('G3', 10000), ('G4', 9000), ('G5', 8000), ('G6', 7000), ('G7', 6000), ('G8', 5000), ('G9', 4000), ('G10', 3000), ('G11', 2000), ('G12', 1000); -- ── employees ──────────────────────────────────────────────── CREATE TABLE employees ( id VARCHAR(20) PRIMARY KEY, username VARCHAR(100) NOT NULL UNIQUE, password_hash VARCHAR(255) NOT NULL, display_name VARCHAR(150) NOT NULL, email VARCHAR(255) NOT NULL UNIQUE, role user_role NOT NULL, manager_level manager_level, job_title VARCHAR(150) NOT NULL, entity VARCHAR(100) NOT NULL, department VARCHAR(100) NOT NULL, grade VARCHAR(5) NOT NULL REFERENCES grade_pay_scales(grade), manager_id VARCHAR(20) REFERENCES employees(id) ON DELETE SET NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); CREATE INDEX idx_employees_role ON employees(role); CREATE INDEX idx_employees_manager_id ON employees(manager_id); CREATE INDEX idx_employees_department ON employees(department); -- ── ot_records ─────────────────────────────────────────────── CREATE TABLE ot_records ( id BIGSERIAL PRIMARY KEY, -- Who & when emp_id VARCHAR(20) NOT NULL REFERENCES employees(id), manager_id VARCHAR(20) NOT NULL REFERENCES employees(id), date DATE NOT NULL, clock_in TIME NOT NULL, clock_out TIME NOT NULL, pre_approved BOOLEAN NOT NULL DEFAULT FALSE, reason TEXT NOT NULL, -- Employee submitted hours sub_regular_ot NUMERIC(5,2) NOT NULL DEFAULT 0, sub_regular_after9pm NUMERIC(5,2) NOT NULL DEFAULT 0, sub_holiday_ot NUMERIC(5,2) NOT NULL DEFAULT 0, sub_total_ot NUMERIC(5,2) NOT NULL, submitted_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), -- L1 approval l1_status approval_status NOT NULL DEFAULT 'Pending', l1_regular_ot NUMERIC(5,2), l1_regular_after9pm NUMERIC(5,2), l1_holiday_ot NUMERIC(5,2), l1_total_ot NUMERIC(5,2), l1_comments TEXT NOT NULL DEFAULT '', l1_approved_by VARCHAR(20) REFERENCES employees(id), l1_actioned_at TIMESTAMPTZ, -- L2 approval l2_status approval_status, l2_comments TEXT NOT NULL DEFAULT '', l2_approved_by VARCHAR(20) REFERENCES employees(id), l2_actioned_at TIMESTAMPTZ, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), UNIQUE (emp_id, date), CONSTRAINT chk_sub_total CHECK ( sub_total_ot = sub_regular_ot + sub_regular_after9pm + sub_holiday_ot ) ); CREATE INDEX idx_ot_emp_date ON ot_records(emp_id, date); CREATE INDEX idx_ot_manager_date ON ot_records(manager_id, date); CREATE INDEX idx_ot_statuses ON ot_records(l1_status, l2_status); CREATE INDEX idx_ot_date ON ot_records(date); -- ── shift_plans ────────────────────────────────────────────── CREATE TABLE shift_plans ( id BIGSERIAL PRIMARY KEY, emp_id VARCHAR(20) NOT NULL REFERENCES employees(id), date DATE NOT NULL, is_workday BOOLEAN NOT NULL, is_public_holiday BOOLEAN NOT NULL DEFAULT FALSE, shift_start TIME, shift_end TIME, ot_start TIME, ot_end TIME, ot_hours NUMERIC(5,2), created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), UNIQUE (emp_id, date) ); CREATE INDEX idx_shift_emp_date ON shift_plans(emp_id, date); -- ── attendance_records ─────────────────────────────────────── CREATE TABLE attendance_records ( id BIGSERIAL PRIMARY KEY, emp_id VARCHAR(20) NOT NULL REFERENCES employees(id), date DATE NOT NULL, status attendance_status NOT NULL, clock_in TIME, clock_out TIME, total_hours NUMERIC(5,2), ot_record_id BIGINT REFERENCES ot_records(id) ON DELETE SET NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), UNIQUE (emp_id, date) ); CREATE INDEX idx_att_emp_date ON attendance_records(emp_id, date); -- ── leave_records ──────────────────────────────────────────── CREATE TABLE leave_records ( id BIGSERIAL PRIMARY KEY, emp_id VARCHAR(20) NOT NULL REFERENCES employees(id), start_date DATE NOT NULL, end_date DATE NOT NULL, leave_type VARCHAR(60) NOT NULL, approved_by VARCHAR(20) REFERENCES employees(id) ON DELETE SET NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), CONSTRAINT chk_leave_dates CHECK (end_date >= start_date) ); CREATE INDEX idx_leave_emp_dates ON leave_records(emp_id, start_date, end_date); -- ── sessions ───────────────────────────────────────────────── CREATE TABLE sessions ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), emp_id VARCHAR(20) NOT NULL REFERENCES employees(id) ON DELETE CASCADE, token_hash VARCHAR(255) NOT NULL UNIQUE, expires_at TIMESTAMPTZ NOT NULL, created_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); CREATE INDEX idx_sessions_token ON sessions(token_hash); CREATE INDEX idx_sessions_expires_at ON sessions(expires_at); -- ── Auto-update updated_at ──────────────────────────────────── CREATE OR REPLACE FUNCTION set_updated_at() RETURNS TRIGGER AS $$ BEGIN NEW.updated_at = NOW(); RETURN NEW; END; $$ LANGUAGE plpgsql; CREATE TRIGGER trg_employees_updated_at BEFORE UPDATE ON employees FOR EACH ROW EXECUTE FUNCTION set_updated_at(); CREATE TRIGGER trg_ot_records_updated_at BEFORE UPDATE ON ot_records FOR EACH ROW EXECUTE FUNCTION set_updated_at();
